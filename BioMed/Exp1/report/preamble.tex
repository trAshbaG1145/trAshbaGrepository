% ---- 页面设置 ----
\usepackage{geometry}
\geometry{a4paper, top=2.4cm, bottom=2.4cm, left=2.4cm, right=2.4cm}
\setlength{\parindent}{2em}
\setlength{\parskip}{0.35em}
\linespread{1.25}

% ---- 字体（Windows 自带） ----
\usepackage{xeCJK}
\setCJKmainfont{SimSun}[BoldFont=SimHei, ItalicFont=KaiTi]
\setCJKsansfont{SimHei}
\setCJKmonofont{SimSun}
\setmainfont{Times New Roman}
\setsansfont{SimHei}
\setmonofont{Consolas}[Scale=0.86]

% ---- 表格 ----
\usepackage{longtable, booktabs, array}
\renewcommand{\arraystretch}{1.25}
\usepackage{float}

% ---- 标题格式 ----
\usepackage{titlesec}
\titleformat{\section}{\sffamily\Large\bfseries}{\thesection}{0.6em}{}
\titleformat{\subsection}{\sffamily\large\bfseries}{\thesubsection}{0.6em}{}
\titleformat{\subsubsection}{\sffamily\normalsize\bfseries}{\thesubsubsection}{0.6em}{}
\titlespacing{\section}{0pt}{1.6em}{0.8em}
\titlespacing{\subsection}{0pt}{1.2em}{0.6em}

% ---- 页眉页脚 ----
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\sffamily 生物医学智能信息处理 · 实验一}
\fancyhead[R]{\small\sffamily 数据降维与特征筛选}
\fancyfoot[C]{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}

% ---- 列表间距 ----
\usepackage{enumitem}
\setlist[itemize]{topsep=0.25em, itemsep=0.15em, parsep=0.1em}
\setlist[enumerate]{topsep=0.25em, itemsep=0.15em, parsep=0.1em}

% ---- 图片：默认不超出页宽 ----
\usepackage{graphicx}
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth\linewidth\else\Gin@nat@width\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth,keepaspectratio}

% ---- 数学 ----
\usepackage{amsmath, amssymb}

% ---- 图题：把自动编号的 "Figure" 改为 "图" ----
\usepackage{caption}
\captionsetup[figure]{
  name=图,
  labelsep=quad,
  font=small,
  justification=centering,
  singlelinecheck=false
}

% ---- 代码块紧凑排版（代码附录动辄上千行，需压缩行距与字号） ----
\usepackage{etoolbox}
\AtBeginEnvironment{Shaded}{%
  \footnotesize
  \linespread{0.90}\selectfont
  \setlength{\parskip}{0pt}%
  \setlength{\topsep}{0.3em}%
}
\AtBeginEnvironment{Highlighting}{%
  \footnotesize
  \linespread{0.90}\selectfont
  \setlength{\parskip}{0pt}%
}
% 缩小代码块内边距
\setlength{\fboxsep}{4pt}
